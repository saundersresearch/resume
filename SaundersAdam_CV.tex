% ---------------------------------------------------------------------------
% Layout machinery: a fixed-width "label" column on the left and a "content"
% column on the right. The label is printed only on the first row of a
% section; every following row/paragraph passes an empty label so the text
% keeps flowing at the content-column indent -- including across page breaks,
% since this is just ordinary paragraphs, not a table. Labels are kept to a
% single line (the column is sized to fit the widest one, "Abstracts/Short
% Papers") so a label never spills onto a second line and leaves a blank gap
% next to it.
% ---------------------------------------------------------------------------
\newlength{\cvlabelwidth}
\setlength{\cvlabelwidth}{1.2in}
\setlist[itemize]{leftmargin=\dimexpr\cvlabelwidth+0.25in\relax,
                  label=\textbullet, itemsep=1pt, topsep=3pt, parsep=0pt,
                  partopsep=0pt}

% \cvlabelbox{content} -- a \cvlabelwidth-wide, top-aligned label box for
% use inline at the start of a \hangindent paragraph. A label too wide
% for \cvlabelwidth (e.g. "Scientific Experience") wraps onto a second
% line inside it -- but a top-aligned multi-line \parbox reports its
% *entire* extra height as depth (everything below its first baseline),
% so without correction the very next line of running text -- whether
% this same row's own wrapped continuation, or the next \cvrow{}{...}
% row's text -- gets pushed down by all of that accumulated depth in one
% lump, i.e. one visible gap exactly where a 2-line label meets whatever
% comes after it. Forcing the depth back down to a normal single line's
% depth (via a reference \strut) doesn't touch what's drawn -- the
% label's later lines still print in full -- it only corrects what TeX
% *thinks* the box's footprint is, so the next line lands at the usual
% single-line spacing instead. A nice side effect: that following text
% now naturally lands flush beside the label's own 2nd/3rd line too,
% since both are back on the document's ordinary line rhythm. \strut
% inside the box also guards against a real oddity: a *completely*
% empty \parbox (the {} label used by every row after the first)
% silently reverts to zero width, which would throw off that row's
% own indent.
\newbox\cvlabelboxreg
\newbox\cvrefboxreg
\newcommand{\cvlabelbox}[1]{%
  \setbox\cvlabelboxreg=\hbox{\parbox[t]{\cvlabelwidth}{\raggedright#1\strut}}%
  \setbox\cvrefboxreg=\hbox{\strut}%
  \dp\cvlabelboxreg=\dp\cvrefboxreg\relax
  \box\cvlabelboxreg}

% \cvrow{label}{text}{date} -- entry row with a right-flushed date
% (the date is kept in an \mbox so it moves to the next line as a whole
% instead of splitting mid-phrase, e.g. "May" / "2023").
\newcommand{\cvrow}[3]{%
  \par\noindent\hangindent=\cvlabelwidth\hangafter=1
  \cvlabelbox{\textbf{#1}}#2\hspace{0.75em}\hfill\mbox{#3}\par}

% \cvpar{label}{text} -- plain running paragraph, no date column
\newcommand{\cvpar}[2]{%
  \par\noindent\hangindent=\cvlabelwidth\hangafter=1
  \cvlabelbox{\textbf{#1}}#2\par}
